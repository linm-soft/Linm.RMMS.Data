  {code:'go-giam-toc', name:'Gờ giảm tốc', geom:'LineString', visible:false},
  {code:'tuong-chan', name:'Tường chắn', geom:'LineString', visible:false},
  {code:'cau', name:'Cầu', geom:'Polygon', visible:false},
  {code:'ho-lan', name:'Hộ lan', geom:'LineString', visible:false},
  {code:'bien-bao', name:'Biển báo', geom:'Point', visible:false},
  {code:'cong', name:'Cống', geom:'Point', visible:false},
  {code:'moc-lo-gioi', name:'Mốc lộ giới', geom:'Point', visible:false},
  {code:'nen-duong', name:'Nền đường', geom:'Polygon', visible:false},
];

let tool = 'select';
let targetLayer = LAYERS[0].code;
let draft = null; // {type, points:[{x,y}], el}
let history = [];
let redoStack = [];
let saved = [];
let zoom = 1;

const svg = document.getElementById('drawSvg');
const statusBar = document.getElementById('statusBar');
const tree = document.getElementById('assetLayers');

function renderTree(){
  tree.innerHTML = LAYERS.map(l => `
    <label class="check">
      <input type="checkbox" data-vis="${l.code}" ${l.visible?'checked':''}>
      <input type="radio" name="target" value="${l.code}" ${l.code===targetLayer?'checked':''}>
      <span>${l.name} <small style="color:#94a3b8">(${l.geom})</small></span>
    </label>`).join('');
}
renderTree();

tree.addEventListener('change', e => {
  const t = e.target;
  if (t.name === 'target') {
    targetLayer = t.value;
    status(`Target: ${layerName(targetLayer)}`);
  }
  if (t.dataset.vis) {
    const L = LAYERS.find(x => x.code === t.dataset.vis);
    if (L) L.visible = t.checked;
  }
});

document.querySelectorAll('input[name=basemap]').forEach(r => {
  r.addEventListener('change', () => {
    document.getElementById('bmLabel').textContent =
      {admin:'Hành chính',traffic:'Giao thông',google:'Google',sat:'Vệ tinh',none:'Không nền'}[r.value];
    const bg = document.getElementById('mapBg');
    if (r.value === 'google' || r.value === 'sat') {
      bg.style.backgroundImage = "url('ref/1-ban-do.png')";
      bg.style.filter = r.value==='sat' ? 'contrast(1.1) saturate(1.2)' : 'saturate(.9)';
    } else if (r.value === 'none') {
      bg.style.backgroundImage = 'none';
      bg.style.background = '#cbd5e1';
    } else {
      bg.style.backgroundImage = "url('ref/1-ban-do.png')";
      bg.style.filter = 'grayscale(.4) brightness(1.05)';
    }
  });
});

document.getElementById('sideTabs').addEventListener('click', e => {
  const b = e.target.closest('button'); if (!b) return;
  document.querySelectorAll('#sideTabs button').forEach(x => x.classList.toggle('on', x===b));
  ['layers','legend','props','results'].forEach(id => {
    document.getElementById('tab-'+id).hidden = b.dataset.tab !== id;
  });

document.querySelector('.toolbar').addEventListener('click', e => {
  const b = e.target.closest('[data-tool]'); if (!b) return;
  const t = b.dataset.tool;
  if (t === 'undo') { undo(); return; }
  if (t === 'redo') { redo(); return; }
  if (t === 'delete') { deleteDraft(); return; }
  if (t === 'zoomIn') { setZoom(zoom*1.1); return; }
  if (t === 'zoomOut') { setZoom(zoom/1.1); return; }
  tool = t;
  document.querySelectorAll('.toolbar .tool[data-tool]').forEach(x => {
    if (['undo','redo','delete','zoomIn','zoomOut'].includes(x.dataset.tool)) return;
    x.classList.toggle('on', x.dataset.tool === tool);
  });
  svg.classList.toggle('pan', tool === 'pan');
  status(toolLabel(tool));
});

document.getElementById('btnZIn').onclick = () => setZoom(zoom*1.1);
document.getElementById('btnZOut').onclick = () => setZoom(zoom/1.1);

function setZoom(z){
  zoom = Math.min(2.2, Math.max(0.7, z));
  document.getElementById('mapBg').style.transform = `scale(${zoom})`;
  document.getElementById('mapBg').style.transformOrigin = 'center center';
}

function toolLabel(t){
  return ({select:'Chế độ chọn',pan:'Pan map',measure:'Đo (mock)',point:'Vẽ điểm — click map',line:'Vẽ đường — click đỉnh, dblclick kết thúc',polygon:'Vẽ vùng — click đỉnh, dblclick đóng',text:'Nhãn (P1 optional)'})[t] || t;
}
function layerName(code){ return (LAYERS.find(l=>l.code===code)||{}).name || code; }
function status(m){ statusBar.textContent = m; }

function svgPoint(evt){
  const r = svg.getBoundingClientRect();
  return { x: evt.clientX - r.left, y: evt.clientY - r.top };
}

svg.addEventListener('click', evt => {
  if (!['point','line','polygon'].includes(tool)) return;
  const p = svgPoint(evt);
  const L = LAYERS.find(l => l.code === targetLayer);
  if (!L) return;

  if (tool === 'point') {
    if (L.geom !== 'Point') {
      status('Loại «'+L.name+'» nên dùng geom '+L.geom+' — vẫn cho phép demo');
    }
    finishPoint(p);
    return;
  }

  if (!draft) {
    draft = { type: tool, points: [p], els: [] };
    addVertex(p);
    status('Đỉnh 1 — tiếp tục click' + (tool!=='point' ? ', dblclick để kết thúc' : ''));
    return;
  }
  draft.points.push(p);
  addVertex(p);
  redrawDraftPath();
  status('Đỉnh '+draft.points.length);
});

svg.addEventListener('dblclick', evt => {
  evt.preventDefault();
  if (!draft || (draft.type !== 'line' && draft.type !== 'polygon')) return;
  if (draft.type === 'line' && draft.points.length < 2) return;
  if (draft.type === 'polygon' && draft.points.length < 3) return;
  finalizeDraft();
});

function addVertex(p){
  const c = document.createElementNS('http://www.w3.org/2000/svg','circle');
  c.setAttribute('cx', p.x); c.setAttribute('cy', p.y);
  c.setAttribute('r', 5); c.setAttribute('fill', '#f59e0b');
  c.setAttribute('stroke', '#fff'); c.setAttribute('stroke-width', '2');
  svg.appendChild(c);
  draft.els.push(c);
}

function redrawDraftPath(){
  if (!draft) return;
  let path = draft.els.find(e => e.tagName === 'polyline' || e.tagName === 'polygon');
  if (path) path.remove();
  const pts = draft.points.map(p => p.x+','+p.y).join(' ');
  if (draft.type === 'line') {
    path = document.createElementNS('http://www.w3.org/2000/svg','polyline');
    path.setAttribute('points', pts);
    path.setAttribute('fill', 'none');
    path.setAttribute('stroke', '#f59e0b');
    path.setAttribute('stroke-width', '3');
    path.setAttribute('stroke-dasharray', '6 4');
  } else {
    path = document.createElementNS('http://www.w3.org/2000/svg','polygon');
    path.setAttribute('points', pts);
    path.setAttribute('fill', 'rgba(245,158,11,.25)');
    path.setAttribute('stroke', '#f59e0b');
    path.setAttribute('stroke-width', '2');
  }
  svg.insertBefore(path, draft.els[0] || null);
  draft.els.unshift(path);
}

function finishPoint(p){
  clearDraftEls();
  const c = document.createElementNS('http://www.w3.org/2000/svg','circle');
  c.setAttribute('cx', p.x); c.setAttribute('cy', p.y);
  c.setAttribute('r', 8); c.setAttribute('fill', '#7c3aed');
  c.setAttribute('stroke', '#fff'); c.setAttribute('stroke-width', '2');
  c.dataset.draft = '1';
  svg.appendChild(c);
  draft = { type:'point', points:[p], els:[c], layer: targetLayer };
  openProps('Point');
  pushHistory();
}

function finalizeDraft(){
  redrawDraftPath();
  draft.layer = targetLayer;
  draft.els.forEach(el => { if (el.tagName!=='circle') { el.setAttribute('stroke','#f59e0b'); }});
  openProps(draft.type === 'line' ? 'LineString' : 'Polygon');
  pushHistory();
  status('Draft xong — nhập thuộc tính và Lưu');
}

function openProps(geomType){
  document.querySelectorAll('#sideTabs button').forEach(x => x.classList.toggle('on', x.dataset.tab==='props'));
  ['layers','legend','props','results'].forEach(id => {
    document.getElementById('tab-'+id).hidden = id !== 'props';
  });
  document.getElementById('fLayer').value = layerName(targetLayer) + ' (' + targetLayer + ')';
  document.getElementById('fGeom').value = geomType + ' · ' + (draft?draft.points.length:0) + ' điểm (demo px)';
  document.getElementById('fCode').value = '';
  document.getElementById('fName').value = '';
  document.getElementById('btnSave').disabled = false;
  document.getElementById('saveMsg').textContent = '';
}

function clearDraftEls(){
  if (!draft) return;
  draft.els.forEach(el => el.remove());
  draft = null;
}

function deleteDraft(){
  clearDraftEls();
  document.getElementById('btnSave').disabled = true;
  status('Đã xoá draft');
}

function pushHistory(){
  history.push(svg.innerHTML);
  if (history.length > 20) history.shift();
  redoStack = [];
}

function undo(){
  if (!history.length) return;
  redoStack.push(svg.innerHTML);
  svg.innerHTML = history.pop() || '';
  draft = null;
  status('Undo');
}
function redo(){
  if (!redoStack.length) return;
  history.push(svg.innerHTML);
  svg.innerHTML = redoStack.pop();
  status('Redo');
}

document.getElementById('btnClearDraft').onclick = deleteDraft;
document.getElementById('btnSave').onclick = () => {
  const code = document.getElementById('fCode').value.trim();
  const name = document.getElementById('fName').value.trim();
  if (!code || !name) {
    document.getElementById('saveMsg').textContent = 'Nhập Mã và Tên.';
    return;
  }
  if (!draft) {
    document.getElementById('saveMsg').textContent = 'Không có draft.';
    return;
  }
  const item = {
    code, name,
    layer: targetLayer,
    geom: document.getElementById('fGeom').value,
    at: new Date().toLocaleTimeString('vi-VN')
  };
  saved.push(item);
  // lock style as saved
  draft.els.forEach(el => {
    if (el.tagName === 'circle' && draft.type==='point') {
      el.setAttribute('fill', '#2563eb');
    } else if (el.getAttribute) {
      el.setAttribute('stroke', '#2563eb');
      el.setAttribute('stroke-dasharray', '');
      if (el.tagName==='polygon') el.setAttribute('fill','rgba(37,99,235,.2)');
    }
  });
  draft = null;
  document.getElementById('btnSave').disabled = true;
  document.getElementById('saveMsg').textContent = 'Đã lưu mock → POST /api/v1/gis/drawings (demo session).';
  renderResults();
  status('Đã lưu '+code);
};

function renderResults(){
  const ul = document.getElementById('resultList');
  ul.innerHTML = saved.length ? saved.map(s =>
    `<li><strong>${s.code}</strong> — ${s.name} · ${layerName(s.layer)} · ${s.at}</li>`
  ).join('') : '<li>Chưa có</li>';
}
