<script>
const LAYERS = [
  {code:'tuyen-duong', name:'Tuyến đường', geom:'LineString'},
  {code:'cot-km', name:'Cột km', geom:'Point'},
  {code:'bien-bao', name:'Biển báo', geom:'Point'},
  {code:'cau', name:'Cầu', geom:'Polygon'},
  {code:'ho-lan', name:'Hộ lan', geom:'LineString'},
  {code:'cong', name:'Cống', geom:'Point'},
  {code:'ta-luy', name:'Ta luy / tường chắn', geom:'Polygon'},
  {code:'mat-duong', name:'Đoạn mặt đường', geom:'LineString'},
];
let target = LAYERS[0].code;
const layersEl = document.getElementById('layers');
layersEl.innerHTML = LAYERS.map((l,i) =>
  `<label class="check"><input type="radio" name="tg" value="${l.code}" ${i===0?'checked':''}> ${l.name} <small style="color:#94a3b8">(${l.geom})</small></label>`
).join('');
layersEl.addEventListener('change', e => {
  if (e.target.name==='tg') {
    target = e.target.value;
    document.getElementById('fLayer').value = layerLabel(target);
  }
});
document.getElementById('fLayer').value = layerLabel(target);
function layerLabel(c){ const L=LAYERS.find(x=>x.code===c); return L?`${L.name} (${L.code})`:c; }

const NA = [18.679, 105.681];
const map = L.map('map', { center: NA, zoom: 11 });

const tiles = {
  osm: L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom:19, attribution:'© OpenStreetMap' }),
  topo: L.tileLayer('https://{s}.tile.opentopomap.org/{z}/{x}/{y}.png', { maxZoom:17, attribution:'© OpenTopoMap' }),
  sat: L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}', { maxZoom:19, attribution:'© Esri' }),
};
let currentTile = tiles.osm.addTo(map);
document.querySelectorAll('input[name=bm]').forEach(r => {
  r.addEventListener('change', () => {
    map.removeLayer(currentTile);
    currentTile = tiles[r.value].addTo(map);
  });
});

const drawn = new L.FeatureGroup().addTo(map);
const draftLayer = new L.FeatureGroup().addTo(map);

const drawControl = new L.Control.Draw({
  position: 'topleft',
  draw: {
    marker: { shapeOptions: {} },
    polyline: { shapeOptions: { color:'#f59e0b', weight:4 } },
    polygon: { allowIntersection:false, shapeOptions: { color:'#f59e0b', fillOpacity:0.25 } },
    rectangle: false,
    circle: false,
    circlemarker: false,
  },
  edit: { featureGroup: drawn, remove: true },
});
map.addControl(drawControl);

let draft = null;
let saved = JSON.parse(localStorage.getItem('rmms-gis-live') || '[]');

map.on(L.Draw.Event.CREATED, e => {
  draftLayer.clearLayers();
  const layer = e.layer;
  draftLayer.addLayer(layer);
  const gj = layer.toGeoJSON();
  draft = { layerCode: target, geometry: gj.geometry, leafletLayer: layer };
  document.getElementById('fLayer').value = layerLabel(target);
  document.getElementById('fGeom').value = JSON.stringify(gj.geometry, null, 2);
  document.getElementById('fCode').value = '';
  document.getElementById('fName').value = '';
  document.getElementById('msg').textContent = 'Draft sẵn — nhập Mã/Tên rồi Lưu local';
});

document.getElementById('btnLocate').onclick = () => map.setView(NA, 11);
document.getElementById('btnClear').onclick = () => {
  draftLayer.clearLayers();
  draft = null;
  document.getElementById('fGeom').value = '';
  document.getElementById('msg').textContent = '';
};
document.getElementById('btnSave').onclick = () => {
  const code = document.getElementById('fCode').value.trim();
  const name = document.getElementById('fName').value.trim();
  if (!draft) { document.getElementById('msg').textContent = 'Chưa có geometry — vẽ trước.'; return; }
  if (!code || !name) { document.getElementById('msg').textContent = 'Nhập Mã và Tên.'; return; }
  const item = {
    id: crypto.randomUUID ? crypto.randomUUID() : String(Date.now()),
    code, name,
    layerCode: draft.layerCode,
    geometry: draft.geometry,
    at: new Date().toISOString(),
  };
  // commit style
  if (draft.leafletLayer.setStyle) {
    draft.leafletLayer.setStyle({ color:'#2563eb', fillColor:'#2563eb', fillOpacity:0.25, weight:3 });
  }
  drawn.addLayer(draft.leafletLayer);
  draft.leafletLayer.bindPopup(`<b>${code}</b><br>${name}<br><small>${draft.layerCode}</small>`);
  draftLayer.clearLayers();
  draft = null;
  saved.push(item);
  localStorage.setItem('rmms-gis-live', JSON.stringify(saved));
  document.getElementById('fGeom').value = '';
  document.getElementById('msg').textContent = 'Đã lưu → mock POST /api/v1/gis/drawings';
  renderSaved();
  // restore from geojson for reload persistence of drawn visuals
  persistDrawnVisual();
};

function persistDrawnVisual(){
  localStorage.setItem('rmms-gis-live-fc', JSON.stringify({
    type:'FeatureCollection',
    features: saved.map(s => ({ type:'Feature', properties:{code:s.code,name:s.name,layerCode:s.layerCode}, geometry:s.geometry }))
  }));
}

function renderSaved(){
  const ul = document.getElementById('saved');
  ul.innerHTML = saved.length
    ? saved.slice().reverse().slice(0,12).map(s => `<li><strong>${s.code}</strong> — ${s.name} · ${s.layerCode}</li>`).join('')
    : '<li>Chưa có</li>';
}

function loadSavedOntoMap(){
  const fc = localStorage.getItem('rmms-gis-live-fc');
  if (!fc) return;
  try {
    const data = JSON.parse(fc);
    L.geoJSON(data, {
      style: { color:'#2563eb', weight:3, fillOpacity:0.25 },
      pointToLayer: (f, latlng) => L.circleMarker(latlng, { radius:8, color:'#fff', fillColor:'#7c3aed', fillOpacity:1, weight:2 }),
      onEachFeature: (f, layer) => {
        const p = f.properties||{};
        layer.bindPopup(`<b>${p.code||''}</b><br>${p.name||''}`);
        drawn.addLayer(layer);
      }
    });
  } catch(e) {}
}

document.getElementById('btnExport').onclick = () => {
  const blob = new Blob([localStorage.getItem('rmms-gis-live-fc') || '{"type":"FeatureCollection","features":[]}'], {type:'application/json'});
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = 'rmms-drawings.geojson';
  a.click();
};
document.getElementById('btnWipe').onclick = () => {
  if (!confirm('Xoá toàn bộ layer đã lưu trên trình duyệt?')) return;
  saved = [];
  localStorage.removeItem('rmms-gis-live');
  localStorage.removeItem('rmms-gis-live-fc');
  drawn.clearLayers();
  draftLayer.clearLayers();
  renderSaved();
};

renderSaved();
loadSavedOntoMap();
</script>
